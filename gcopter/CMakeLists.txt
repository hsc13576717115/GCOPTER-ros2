cmake_minimum_required(VERSION 3.16)
project(gcopter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(ompl REQUIRED)
add_executable(global_planning src/global_planning.cpp)
target_include_directories(global_planning PRIVATE include ${OMPL_INCLUDE_DIRS})
target_link_libraries(global_planning Eigen3::Eigen ${OMPL_LIBRARIES})
ament_target_dependencies(global_planning rclcpp std_msgs geometry_msgs sensor_msgs visualization_msgs)
install(TARGETS global_planning DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(Eigen3 ompl)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_geometry test/test_geometry.cpp)
  target_include_directories(test_geometry PRIVATE include ${OMPL_INCLUDE_DIRS})
  target_link_libraries(test_geometry Eigen3::Eigen ${OMPL_LIBRARIES})
  ament_add_pytest_test(test_demo test/test_demo.py TIMEOUT 240)
endif()
install(PROGRAMS scripts/verify_demo.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
